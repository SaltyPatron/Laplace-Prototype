cmake_minimum_required(VERSION 3.24)
project(laplace_proofs C)
set(CMAKE_C_STANDARD 11)
find_package(PostgreSQL REQUIRED)
find_path(LAPLACE_INCLUDE_DIR laplace/laplace.h REQUIRED)
find_library(LAPLACE_LIBRARY NAMES laplace laplace_native REQUIRED)
get_filename_component(LAPLACE_LIBRARY_DIR "${LAPLACE_LIBRARY}" DIRECTORY)
find_library(LAPLACE_TEXT_LIBRARY NAMES laplace_text HINTS "${LAPLACE_LIBRARY_DIR}" REQUIRED)
find_package(ICU REQUIRED COMPONENTS uc i18n)
add_executable(proof_native proof_native.c)
target_include_directories(proof_native PRIVATE "${LAPLACE_INCLUDE_DIR}")
target_link_libraries(proof_native PRIVATE "${LAPLACE_TEXT_LIBRARY}" "${LAPLACE_LIBRARY}" PostgreSQL::PostgreSQL ICU::uc ICU::i18n)
if(MSVC)
  target_compile_definitions(proof_native PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(proof_native PRIVATE /fp:precise)
else()
  target_compile_options(proof_native PRIVATE -Wall -Wextra -ffp-contract=off -fno-fast-math)
endif()
