cmake_minimum_required(VERSION 3.20)
project(laplace_model_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(LLAMA_SOURCE_DIR "" CACHE PATH "CPU llama.cpp checkout")
set(LLAMA_LIBRARY_DIR "" CACHE PATH "CPU llama.cpp shared libraries")
set(LAPLACE_SOURCE_DIR "" CACHE PATH "Laplace monorepo checkout")
set(LAPLACE_SYNTHESIS_DIR "" CACHE PATH "Built liblaplace_synthesis directory")
set(LAPLACE_RUNTIME_DIR "/opt/laplace/lib" CACHE PATH "Other Laplace runtime dependencies")
option(LAPLACE_COMPONENT_ONLY "Build the connected model-component calculation" OFF)
if(LAPLACE_COMPONENT_ONLY)
  find_library(DYNAMICS_LIBRARY NAMES laplace_dynamics PATHS ${LAPLACE_RUNTIME_DIR} NO_DEFAULT_PATH REQUIRED)
  find_library(CORE_LIBRARY NAMES laplace_core PATHS ${LAPLACE_RUNTIME_DIR} NO_DEFAULT_PATH REQUIRED)
  add_library(laplace-model-components SHARED model-components.cpp model-records.cpp)
  target_include_directories(laplace-model-components PRIVATE ${LAPLACE_SOURCE_DIR}/engine/dynamics/include
    ${LAPLACE_SOURCE_DIR}/engine/core/include)
  target_link_libraries(laplace-model-components PRIVATE ${DYNAMICS_LIBRARY} ${CORE_LIBRARY})
  target_compile_options(laplace-model-components PRIVATE -O3 -mf16c -ffp-contract=off -fno-fast-math)
  set_target_properties(laplace-model-components PROPERTIES BUILD_RPATH "${LAPLACE_RUNTIME_DIR}")
  return()
endif()
foreach(required LLAMA_SOURCE_DIR LLAMA_LIBRARY_DIR LAPLACE_SOURCE_DIR LAPLACE_SYNTHESIS_DIR)
  if(NOT ${required})
    message(FATAL_ERROR "${required} is required")
  endif()
endforeach()
add_executable(laplace-readout-calibrate readout-calibrate.cpp)
target_include_directories(laplace-readout-calibrate PRIVATE
  ${LLAMA_SOURCE_DIR}/include ${LLAMA_SOURCE_DIR}/ggml/include ${LAPLACE_SOURCE_DIR}/engine/synthesis/include)
target_compile_options(laplace-readout-calibrate PRIVATE -ffp-contract=off -fno-fast-math)
# Some moved llama.cpp builds retain an obsolete library RUNPATH. Link every CPU
# dependency directly so their explicit selected directory resolves each SONAME.
target_link_options(laplace-readout-calibrate PRIVATE -Wl,--no-as-needed)
foreach(component llama ggml ggml-cpu ggml-base)
  find_library(${component}_LIBRARY NAMES ${component} PATHS ${LLAMA_LIBRARY_DIR} NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(laplace-readout-calibrate PRIVATE ${${component}_LIBRARY})
endforeach()
find_library(SYNTHESIS_LIBRARY NAMES laplace_synthesis PATHS ${LAPLACE_SYNTHESIS_DIR} NO_DEFAULT_PATH REQUIRED)
target_link_libraries(laplace-readout-calibrate PRIVATE ${SYNTHESIS_LIBRARY})
set_target_properties(laplace-readout-calibrate PROPERTIES
  BUILD_RPATH "${LLAMA_LIBRARY_DIR};${LAPLACE_SYNTHESIS_DIR};${LAPLACE_RUNTIME_DIR}")

add_executable(laplace-verify-native-states verify-native-states.cpp)
target_include_directories(laplace-verify-native-states PRIVATE ${LLAMA_SOURCE_DIR}/include ${LLAMA_SOURCE_DIR}/ggml/include)
target_link_options(laplace-verify-native-states PRIVATE -Wl,--no-as-needed)
foreach(component llama ggml ggml-cpu ggml-base)
  target_link_libraries(laplace-verify-native-states PRIVATE ${${component}_LIBRARY})
endforeach()
set_target_properties(laplace-verify-native-states PROPERTIES BUILD_RPATH "${LLAMA_LIBRARY_DIR}")

add_executable(laplace-safetensors-consumer safetensors-consumer.cpp)
target_include_directories(laplace-safetensors-consumer PRIVATE ${LLAMA_SOURCE_DIR}/ggml/include ${LAPLACE_SOURCE_DIR}/engine/synthesis/include)
target_link_options(laplace-safetensors-consumer PRIVATE -Wl,--no-as-needed)
foreach(component llama ggml ggml-cpu ggml-base)
  target_link_libraries(laplace-safetensors-consumer PRIVATE ${${component}_LIBRARY})
endforeach()
target_link_libraries(laplace-safetensors-consumer PRIVATE ${SYNTHESIS_LIBRARY})
set_target_properties(laplace-safetensors-consumer PROPERTIES BUILD_RPATH "${LLAMA_LIBRARY_DIR};${LAPLACE_SYNTHESIS_DIR};${LAPLACE_RUNTIME_DIR}")
